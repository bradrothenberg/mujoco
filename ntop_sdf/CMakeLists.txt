cmake_minimum_required(VERSION 3.20)
project(ntop_sdf LANGUAGES C CXX)

add_library(ntop_sdf STATIC
  src/field.cpp src/world.cpp
  vendor/giles/collide.cpp vendor/giles/optimize.cpp vendor/giles/solver.cpp
  vendor/giles/shapes.cpp vendor/giles/mesh.cpp)
target_compile_features(ntop_sdf PUBLIC cxx_std_20)
target_include_directories(ntop_sdf PUBLIC include PRIVATE vendor/giles)

set(NTOP_CORE_ROOT "" CACHE PATH "Optional nTop Core SDK directory")
if(NTOP_CORE_ROOT)
  find_path(NTOP_CORE_INCLUDE_DIR ntop_core/ntop_core.h PATHS "${NTOP_CORE_ROOT}/include" NO_DEFAULT_PATH REQUIRED)
  find_library(NTOP_CORE_LIBRARY NAMES ntop_core PATHS "${NTOP_CORE_ROOT}/lib" NO_DEFAULT_PATH REQUIRED)
  add_library(ntop_sdf_core STATIC src/ntop_core.cpp)
  target_compile_features(ntop_sdf_core PUBLIC cxx_std_20)
  target_include_directories(ntop_sdf_core PRIVATE "${NTOP_CORE_INCLUDE_DIR}")
  target_link_libraries(ntop_sdf_core PUBLIC ntop_sdf "${NTOP_CORE_LIBRARY}")
  add_executable(ntop_core_info tools/ntop_core_info.cpp)
  target_link_libraries(ntop_core_info PRIVATE ntop_sdf_core)
  if(WIN32)
    find_file(NTOP_CORE_RUNTIME NAMES ntop_core.dll PATHS "${NTOP_CORE_ROOT}/lib" NO_DEFAULT_PATH REQUIRED)
    configure_file("${NTOP_CORE_RUNTIME}" "${CMAKE_CURRENT_BINARY_DIR}/ntop_core.dll" COPYONLY)
    target_link_libraries(ntop_core_info PRIVATE shell32)
  endif()
endif()

add_executable(ntop_sdf_bench bench/benchmark.cpp)
target_link_libraries(ntop_sdf_bench PRIVATE ntop_sdf)
add_executable(ntop_sdf_accuracy bench/contact_accuracy.cpp)
target_link_libraries(ntop_sdf_accuracy PRIVATE ntop_sdf)
if(WIN32)
  target_link_libraries(ntop_sdf_bench PRIVATE psapi)
endif()

option(NTOP_SDF_BUILD_DEMO "Build the interactive implicit collision viewer" OFF)
set(NTOP_SDF_REFERENCE_SOURCE "" CACHE PATH "Optional sdf-dcd-demo checkout with SDL and ImGui submodules")
if(NTOP_SDF_BUILD_DEMO)
  if(NTOP_SDF_REFERENCE_SOURCE)
    set(SDL_SOURCE_DIR "${NTOP_SDF_REFERENCE_SOURCE}/external/SDL")
    set(IMGUI_SOURCE_DIR "${NTOP_SDF_REFERENCE_SOURCE}/external/imgui")
  else()
    include(FetchContent)
    FetchContent_Declare(demo_sdl GIT_REPOSITORY https://github.com/libsdl-org/SDL.git GIT_TAG 98d1f3a45aae568ccd6ed5fec179330f47d4d356)
    FetchContent_Declare(demo_imgui GIT_REPOSITORY https://github.com/ocornut/imgui.git GIT_TAG f5befd2d29e66809cd1110a152e375a7f1981f06)
    FetchContent_GetProperties(demo_sdl)
    if(NOT demo_sdl_POPULATED)
      FetchContent_Populate(demo_sdl)
    endif()
    FetchContent_GetProperties(demo_imgui)
    if(NOT demo_imgui_POPULATED)
      FetchContent_Populate(demo_imgui)
    endif()
    set(SDL_SOURCE_DIR "${demo_sdl_SOURCE_DIR}")
    set(IMGUI_SOURCE_DIR "${demo_imgui_SOURCE_DIR}")
  endif()
  set(SDL_SHARED OFF CACHE BOOL "" FORCE)
  set(SDL_STATIC ON CACHE BOOL "" FORCE)
  set(SDL_TEST OFF CACHE BOOL "" FORCE)
  add_subdirectory("${SDL_SOURCE_DIR}" "${CMAKE_CURRENT_BINARY_DIR}/sdl" EXCLUDE_FROM_ALL)
  find_package(OpenGL REQUIRED)
  add_executable(ntop_sdf_demo
    demo/main.cpp demo/scenes.cpp demo/gl.cpp demo/render.cpp
    "${IMGUI_SOURCE_DIR}/imgui.cpp" "${IMGUI_SOURCE_DIR}/imgui_draw.cpp"
    "${IMGUI_SOURCE_DIR}/imgui_tables.cpp" "${IMGUI_SOURCE_DIR}/imgui_widgets.cpp"
    "${IMGUI_SOURCE_DIR}/backends/imgui_impl_sdl2.cpp" "${IMGUI_SOURCE_DIR}/backends/imgui_impl_opengl3.cpp")
  target_include_directories(ntop_sdf_demo PRIVATE vendor/giles "${IMGUI_SOURCE_DIR}" "${IMGUI_SOURCE_DIR}/backends")
  target_link_libraries(ntop_sdf_demo PRIVATE ntop_sdf SDL2::SDL2main SDL2::SDL2-static OpenGL::GL)
  if(NTOP_CORE_ROOT)
    target_sources(ntop_sdf_demo PRIVATE demo/ntop_import.cpp)
    target_compile_definitions(ntop_sdf_demo PRIVATE NTOP_SDF_WITH_CORE=1)
    target_include_directories(ntop_sdf_demo PRIVATE "${NTOP_CORE_INCLUDE_DIR}")
    target_link_libraries(ntop_sdf_demo PRIVATE ntop_sdf_core)
    if(WIN32)
      target_link_libraries(ntop_sdf_demo PRIVATE shell32)
    endif()
  endif()
  if(WIN32)
    target_compile_definitions(ntop_sdf_demo PRIVATE NOMINMAX WIN32_LEAN_AND_MEAN _CRT_SECURE_NO_WARNINGS)
    target_link_libraries(ntop_sdf_demo PRIVATE comdlg32)
  endif()
endif()

include(CTest)
if(BUILD_TESTING)
  add_executable(ntop_sdf_tests tests/tests.cpp)
  target_link_libraries(ntop_sdf_tests PRIVATE ntop_sdf)
  add_test(NAME ntop_sdf_tests COMMAND ntop_sdf_tests)
  add_test(NAME ntop_sdf_accuracy COMMAND ntop_sdf_accuracy)
  if(NTOP_CORE_ROOT)
    add_executable(ntop_core_test tests/ntop_core_test.cpp)
    target_link_libraries(ntop_core_test PRIVATE ntop_sdf_core)
    if(WIN32)
      target_link_libraries(ntop_core_test PRIVATE shell32)
    endif()
    set(NTOP_CORE_TEST_IMPLICIT "${NTOP_CORE_ROOT}/examples/assets/sphere_5mm_radius.implicit" CACHE FILEPATH "nTop Core sphere fixture")
    if(EXISTS "${NTOP_CORE_TEST_IMPLICIT}")
      add_test(NAME ntop_core_test COMMAND ntop_core_test "${NTOP_CORE_TEST_IMPLICIT}")
      if(WIN32)
        configure_file("${NTOP_CORE_TEST_IMPLICIT}" "${CMAKE_CURRENT_BINARY_DIR}/nTop 测试.implicit" COPYONLY)
        add_test(NAME ntop_core_unicode_cli_test COMMAND ntop_core_info "${CMAKE_CURRENT_BINARY_DIR}/nTop 测试.implicit")
      endif()
    else()
      message(WARNING "nTop Core sphere fixture not found; set NTOP_CORE_TEST_IMPLICIT to run integration tests")
    endif()
  endif()
endif()

set(MUJOCO_BUILD_DIR "" CACHE PATH "Optional MuJoCo build tree")
if(MUJOCO_BUILD_DIR)
  find_library(MUJOCO_LIBRARY NAMES mujoco
    PATHS "${MUJOCO_BUILD_DIR}/lib" "${MUJOCO_BUILD_DIR}/lib/Release" "${MUJOCO_BUILD_DIR}/Release"
    NO_DEFAULT_PATH REQUIRED)
  add_library(ntop_mujoco STATIC src/mujoco_plugin.cpp src/mujoco_collision.cpp)
  target_compile_features(ntop_mujoco PUBLIC cxx_std_20)
  target_include_directories(ntop_mujoco PRIVATE "${CMAKE_CURRENT_SOURCE_DIR}/../include" vendor/giles)
  target_link_libraries(ntop_mujoco PUBLIC ntop_sdf "${MUJOCO_LIBRARY}")
  add_executable(mujoco_bench bench/mujoco_benchmark.cpp)
  target_include_directories(mujoco_bench PRIVATE "${CMAKE_CURRENT_SOURCE_DIR}/../include")
  target_link_libraries(mujoco_bench PRIVATE ntop_mujoco)
  if(NTOP_CORE_ROOT)
    target_compile_definitions(mujoco_bench PRIVATE NTOP_SDF_WITH_CORE=1)
    target_link_libraries(mujoco_bench PRIVATE ntop_sdf_core)
    add_executable(ntop_mujoco_run tools/ntop_mujoco_run.cpp)
    target_include_directories(ntop_mujoco_run PRIVATE "${CMAKE_CURRENT_SOURCE_DIR}/../include")
    target_link_libraries(ntop_mujoco_run PRIVATE ntop_mujoco ntop_sdf_core)
    if(WIN32)
      target_link_libraries(mujoco_bench PRIVATE shell32)
      target_link_libraries(ntop_mujoco_run PRIVATE shell32)
    endif()
  endif()
  if(WIN32)
    target_link_libraries(mujoco_bench PRIVATE psapi)
  endif()
  if(WIN32)
    find_file(MUJOCO_RUNTIME NAMES libmujoco.dll mujoco.dll
      PATHS "${MUJOCO_BUILD_DIR}/bin" "${MUJOCO_BUILD_DIR}/bin/Release" "${MUJOCO_BUILD_DIR}/Release"
      NO_DEFAULT_PATH REQUIRED)
    get_filename_component(MUJOCO_RUNTIME_NAME "${MUJOCO_RUNTIME}" NAME)
    configure_file("${MUJOCO_RUNTIME}" "${CMAKE_CURRENT_BINARY_DIR}/${MUJOCO_RUNTIME_NAME}" COPYONLY)
  endif()
  if(BUILD_TESTING)
    add_executable(mujoco_plugin_test tests/mujoco_plugin_test.cpp)
    target_include_directories(mujoco_plugin_test PRIVATE "${CMAKE_CURRENT_SOURCE_DIR}/../include")
    target_link_libraries(mujoco_plugin_test PRIVATE ntop_mujoco)
    add_test(NAME mujoco_plugin_test COMMAND mujoco_plugin_test)
    if(NTOP_CORE_ROOT)
      add_executable(ntop_core_mujoco_test tests/ntop_core_mujoco_test.cpp)
      target_include_directories(ntop_core_mujoco_test PRIVATE "${CMAKE_CURRENT_SOURCE_DIR}/../include")
      target_link_libraries(ntop_core_mujoco_test PRIVATE ntop_mujoco ntop_sdf_core)
      if(WIN32)
        target_link_libraries(ntop_core_mujoco_test PRIVATE shell32)
      endif()
      if(EXISTS "${NTOP_CORE_TEST_IMPLICIT}")
        add_test(NAME ntop_core_mujoco_test COMMAND ntop_core_mujoco_test "${NTOP_CORE_TEST_IMPLICIT}")
        add_test(NAME ntop_core_offset_mujoco_test COMMAND ntop_core_mujoco_test "${NTOP_CORE_TEST_IMPLICIT}" offset)
        add_test(NAME ntop_core_rotated_mujoco_test COMMAND ntop_core_mujoco_test "${NTOP_CORE_TEST_IMPLICIT}" offset_rotated)
        add_test(NAME ntop_mujoco_run_test COMMAND ntop_mujoco_run "${NTOP_CORE_TEST_IMPLICIT}" 1000 0.25 240)
        set_tests_properties(ntop_mujoco_run_test PROPERTIES PASS_REGULAR_EXPRESSION "total_contacts=[1-9][0-9]*")
        if(WIN32)
          add_test(NAME ntop_mujoco_unicode_cli_test COMMAND ntop_mujoco_run "${CMAKE_CURRENT_BINARY_DIR}/nTop 测试.implicit" 1000 0.25 240)
          set_tests_properties(ntop_mujoco_unicode_cli_test PROPERTIES PASS_REGULAR_EXPRESSION "total_contacts=[1-9][0-9]*")
        endif()
      endif()
      set(NTOP_CORE_GYROID_IMPLICIT "${NTOP_CORE_ROOT}/examples/assets/gyroid_sphere_5mm_radius.implicit" CACHE FILEPATH "nTop Core gyroid fixture")
      if(EXISTS "${NTOP_CORE_GYROID_IMPLICIT}")
        add_test(NAME ntop_core_gyroid_mujoco_test COMMAND ntop_core_mujoco_test "${NTOP_CORE_GYROID_IMPLICIT}" gyroid)
        add_test(NAME ntop_mujoco_gyroid_run_test COMMAND ntop_mujoco_run "${NTOP_CORE_GYROID_IMPLICIT}" 1000 0.25 240)
        set_tests_properties(ntop_mujoco_gyroid_run_test PROPERTIES PASS_REGULAR_EXPRESSION "total_contacts=[1-9][0-9]*")
      endif()
    endif()
  endif()
endif()
